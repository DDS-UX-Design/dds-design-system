export const parseBacklog = (source) => {
  const sections = [];
  let area = '';
  let component = '';

source.split('\n').forEach((line) => {
if (line.startsWith('## ')) {
area = line.slice(3).trim();
component = '';
return;
}

    if (line.startsWith('### ')) {
      component = line.slice(4).trim();
      return;
    }

    const reference = line.match(/^Figma reference: \[(.*?)\]\((.*?)\)$/);

    if (reference && component) {
      const sectionName = `${area} / ${component}`;
      let section = sections.find((item) => item.name === sectionName);

      if (!section) {
        section = { name: sectionName, tasks: [] };
        sections.push(section);
      }

      section.reference = { label: reference[1], href: reference[2] };
      return;
    }

    const match = line.match(/^- \[([ xX])\] \*\*(.*?)\*\* — (.*)$/);
    if (!match || area === 'Future maintenance') return;

    const sectionName = component ? `${area} / ${component}` : area;
    let section = sections.find((item) => item.name === sectionName);

    if (!section) {
      section = { name: sectionName, tasks: [] };
      sections.push(section);
    }

    const [priority = 'P3', classification = 'Design review'] = match[2]
      .split('·')
      .map((part) => part.trim());
    const text = match[3].trim();

    section.tasks.push({
      id: `${sectionName}:${text}`,
      priority,
      classification,
      text,
      sourceCompleted: match[1].toLowerCase() === 'x',
    });

});

return sections;
};

export const BacklogPage = () => {
  const sections = useMemo(
    () => parseBacklog(`${backlog}\n${checkboxBacklog}\n${dateInputBacklog}`),
    [],
  );
  const tasks = useMemo(() => sections.flatMap((section) => section.tasks), [sections]);
  const [completed, setCompleted] = useState(() => {
    if (typeof window === 'undefined') return {};

    try {
      return JSON.parse(window.localStorage.getItem('dds-figma-backlog') || '{}');
    } catch {
      return {};
    }

});

useEffect(() => {
window.localStorage.setItem('dds-figma-backlog', JSON.stringify(completed));
}, [completed]);

const isComplete = (task) => task.sourceCompleted || Boolean(completed[task.id]);
const completedCount = tasks.filter(isComplete).length;

const toggle = (task) => {
if (task.sourceCompleted) return;
setCompleted((current) => ({ ...current, [task.id]: !current[task.id] }));
};

const clearLocalProgress = () => setCompleted({});

return (

<main className="figma-backlog">
<header className="figma-backlog__header">
<div>
<p className="figma-backlog__eyebrow">DDS DESIGN SYSTEM</p>
<h1>Figma design backlog</h1>
<p>
Design gaps and unresolved decisions found while comparing the DDS Design System in Figma with the coded
components and Storybook documentation.
</p>
</div>
<div className="figma-backlog__progress" aria-label={`${completedCount} of ${tasks.length} enhancements complete`}>
<strong>{completedCount}/{tasks.length}</strong>
<span>complete</span>
</div>
</header>

      <aside className="figma-backlog__notice">
        <span>
          Checked items are saved in this browser. Update this Storybook backlog source to record
          completion for the whole team. Storybook source: <strong>{backlogRevision}</strong>.
        </span>
        <button type="button" onClick={clearLocalProgress}>Reset</button>
      </aside>

      <div className="figma-backlog__legend" aria-label="Priority legend">
        <span><b className="priority priority--p0">P0</b> Blocks safe use</span>
        <span><b className="priority priority--p1">P1</b> Important behavior</span>
        <span><b className="priority priority--p2">P2</b> Completeness</span>
        <span><b className="priority priority--p3">P3</b> Guidance or polish</span>
      </div>

      <div className="figma-backlog__sections">
        {sections.map((section) => {
          const sectionComplete = section.tasks.filter(isComplete).length;

          return (
            <section className="figma-backlog__section" key={section.name}>
              <div className="figma-backlog__section-heading">
                <div>
                  <h2>{section.name}</h2>
                  {section.reference && (
                    <a href={section.reference.href} target="_blank" rel="noreferrer">
                      View {section.reference.label} in Figma
                    </a>
                  )}
                </div>
                <span>{sectionComplete}/{section.tasks.length}</span>
              </div>
              <ul>
                {section.tasks.map((task) => {
                  const checked = isComplete(task);

                  return (
                    <li className={checked ? 'is-complete' : ''} key={task.id}>
                      <label>
                        <input
                          type="checkbox"
                          checked={checked}
                          disabled={task.sourceCompleted}
                          onChange={() => toggle(task)}
                        />
                        <span className={`priority priority--${task.priority.toLowerCase()}`}>
                          {task.priority}
                        </span>
                        <span className="figma-backlog__task-copy">
                          <strong>{task.classification}</strong>
                          <span>{task.text}</span>
                        </span>
                      </label>
                    </li>
                  );
                })}
              </ul>
            </section>
          );
        })}
      </div>
    </main>

);
};
